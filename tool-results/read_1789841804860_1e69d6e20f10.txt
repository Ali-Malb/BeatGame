   263→        float hash(vec3 p) {

   265→          p *= 17.0;
   266→          return fract(p.x * p.y * p.z * (p.x + p.y + p.z));
   267→        }

   269→          vec3 d = normalize(vDir);
   270→          float h = clamp(d.y, -1.0, 1.0);
   271→          vec3 col = mix(horizon, mid, smoothstep(-0.02, 0.28, h));
   272→          col = mix(col, zenith, smoothstep(0.22, 0.75, h));
   273→          // sun / moon

   478→  /** main per-frame update */
   479→  update(
   480→    dt: number,
   481→    playerPos: THREE.Vector3,
   482→    playerForward: THREE.Vector3,
   483→    playerVel: THREE.Vector3,
   484→    bike: Motorcycle,
   485→    traffic: TrafficManager,
   486→    highway: Highway,
   487→    camera: THREE.Camera
   488→  ) {
   489→    this.time += dt;
   490→
   491→    // auto cycle
   492→    if (this.autoCycle) {
   493→      this.autoTimer += dt;
   494→      if (this.autoTimer > 75) {
   495→        this.autoTimer = 0;
   496→        this.setPreset((this.presetIndex + 1) % 4);
   497→      }
   498→    }
   499→
   500→    // transition blending
   501→    if (this.transitionT < 1) {
   502→      this.transitionT = Math.min(1, this.transitionT + dt / 2.5);
   503→      const t = this.transitionT;
   504→      blendParams(this.current, this.target, t);
   505→      this.applyAll();
   506→    }
   507→
   508→    // sky follows camera
   509→    this.sky.position.copy(camera.position);
   510→    this.skyMat.uniforms.time.value = this.time;
   511→
   512→    // sun light placement (relative to player so shadows follow)
   513→    const sd = this.current.sunDir;
   514→    this.sun.position.set(playerPos.x + sd.x * 180, playerPos.y + sd.y * 180, playerPos.z + sd.z * 180);
   515→    this.sun.target.position.copy(playerPos);
   516→    this.sun.target.updateMatrixWorld();
   517→    this.sunGlare.position.set(playerPos.x + sd.x * 2300, playerPos.y + sd.y * 2300, playerPos.z + sd.z * 2300);
   518→    (this.sunGlare.material as THREE.SpriteMaterial).opacity = clamp(this.current.glare, 0, 1);
   519→    (this.sunGlare.material as THREE.SpriteMaterial).color.copy(this.current.sunColor);
   520→
   521→    // traffic headlight cones + wet asphalt + lamp emission (cheap global tweaks)
   522→    traffic.headlightsOn = this.headlightsOn;
   523→    const wet = this.current.wetness;
   524→    this.mats.asphalt.roughness = lerp(0.93, 0.12, wet);
   525→    this.mats.asphalt.metalness = lerp(0.02, 0.62, wet);
   526→    this.mats.asphalt.envMapIntensity = lerp(0.25, 1.6, wet);
   527→    this.mats.asphaltOncoming.roughness = lerp(0.95, 0.15, wet);
   528→    this.mats.asphaltOncoming.metalness = lerp(0.0, 0.5, wet);
   529→    this.mats.asphaltOncoming.envMapIntensity = lerp(0.2, 1.2, wet);
   530→
   531→    // env refresh: only while the sky is transitioning, plus a rare top-up
   532→    this.envRefreshTimer += dt;
   533→    if (this.envRefreshTimer > 0.8 && (this.transitionT < 1 || this.envRefreshTimer > 30)) {
   534→      this.envRefreshTimer = 0;
   535→      this.refreshEnv();
   536→    }
   537→
   538→    // bike headlight
   539→    bike.joints.headlightSpot.intensity = damp(bike.joints.headlightSpot.intensity, this.current.headlights, 3, dt);
   540→    bike.joints.headlightSpot.visible = this.current.headlights > 1;
   541→
   542→    // -------- rain particles --------
   543→    const rainAmt = this.current.rain;
   544→    this.rainLines.visible = rainAmt > 0.02;
   545→    if (this.rainLines.visible) {
   546→      this.updateRain(dt, camera, playerVel, rainAmt);
   547→    }
   548→
   549→    // -------- embers --------
   550→    this.embers.visible = this.current.embers > 0.02;
   551→    if (this.embers.visible) {
   552→      this.updateEmbers(dt, playerPos, playerForward);
   553→    }
   554→
   555→    // -------- tire spray --------
   556→    this.updateSpray(dt, bike, traffic, highway, rainAmt);
   557→  }

   559→  private updateRain(dt: number, camera: THREE.Camera, playerVel: THREE.Vector3, amount: number) {
   560→    const camPos = camera.position;
   561→    // camera-space-ish wind: rain falls and streaks backward relative to motion
   562→    const back = playerVel.clone().multiplyScalar(-0.75);
   563→    const velX = back.x * 0.2;
   564→    const velY = -21;
   565→    const velZ = back.z * 0.2;
   566→    const streak = 0.5 + playerVel.length() * 0.012;
   567→    for (let i = 0; i < this.rainCount; i++) {
   568→      let x = this.rainPos[i * 6];
   569→      let y = this.rainPos[i * 6 + 1];
   570→      let z = this.rainPos[i * 6 + 2];
   571→      x += velX * dt;
   572→      y += velY * dt;
   573→      z += velZ * dt;
   574→      // wrap in a 60×26×90 box around the camera
   575→      if (y < camPos.y - 6) y += 26;
   576→      if (y > camPos.y + 20) y -= 26;
   577→      if (x < camPos.x - 30) x += 60;
   578→      if (x > camPos.x + 30) x -= 60;
   579→      if (z < camPos.z - 45) z += 90;
   580→      if (z > camPos.z + 45) z -= 90;
   581→      this.rainPos[i * 6] = x;
   582→      this.rainPos[i * 6 + 1] = y;
   583→      this.rainPos[i * 6 + 2] = z;
   584→      // tail endpoint = velocity direction
   585→      const vl = Math.hypot(velX, velY, velZ) + 1;
   586→      this.rainPos[i * 6 + 3] = x - (velX / vl) * streak;
   587→      this.rainPos[i * 6 + 4] = y - (velY / vl) * streak;
   588→      this.rainPos[i * 6 + 5] = z - (velZ / vl) * streak;
   589→    }
   590→    (this.rainLines.geometry.attributes.position as THREE.BufferAttribute).needsUpdate = true;
   591→    (this.rainLines.material as THREE.LineBasicMaterial).opacity = 0.3 * amount;
   592→    // seed any zeroed drops
   593→    for (let i = 0; i < this.rainCount; i++) {
   594→      if (this.rainPos[i * 6] === 0 && this.rainPos[i * 6 + 1] === 0 && this.rainPos[i * 6 + 2] === 0) {
   595→        this.rainPos[i * 6] = camPos.x + this.rng.range(-28, 28);
   596→        this.rainPos[i * 6 + 1] = camPos.y + this.rng.range(-4, 18);
   597→        this.rainPos[i * 6 + 2] = camPos.z + this.rng.range(-40, 40);
   598→      }
   599→    }
   600→  }

   602→  private updateEmbers(dt: number, playerPos: THREE.Vector3, playerForward: THREE.Vector3) {
   603→    const drift = 2.2;
   604→    for (let i = 0; i < this.emberCount; i++) {
   605→      const spd = this.emberMeta[i * 2];
   606→      const phase = this.emberMeta[i * 2 + 1];
   607→      this.emberPos[i * 3] += (drift + Math.sin(this.time * 0.7 + phase) * 0.8) * spd * dt;
   608→      this.emberPos[i * 3 + 1] += Math.sin(this.time * 1.3 + phase * 2) * 0.35 * dt + 0.05 * dt;
   609→      this.emberPos[i * 3 + 2] += Math.cos(this.time * 0.5 + phase) * 0.5 * spd * dt;
   610→      // wrap around player bubble (roughly aligned with travel)
   611→      const dx = this.emberPos[i * 3] - playerPos.x;
   612→      const dz = this.emberPos[i * 3 + 2] - playerPos.z;
   613→      const fwd = playerForward;
   614→      const along = dx * fwd.x + dz * fwd.z;
   615→      const lat = dx * fwd.z - dz * fwd.x;
   616→      if (along < -60) {
   617→        this.emberPos[i * 3 + 2] += 200 * Math.sign(fwd.z || 1);
   618→        this.emberPos[i * 3] += 200 * Math.sign(fwd.x || 1);
   619→      }
   620→      if (Math.abs(lat) > 32) {
   621→        this.emberPos[i * 3] -= Math.sign(lat) * 0.4;
   622→      }
   623→      if (this.emberPos[i * 3 + 1] > 12) this.emberPos[i * 3 + 1] = 0.3;
   624→    }
   625→    (this.embers.geometry.attributes.position as THREE.BufferAttribute).needsUpdate = true;
   626→  }

   628→  private updateSpray(dt: number, bike: Motorcycle, traffic: TrafficManager, highway: Highway, rainAmt: number) {
   629→    if (rainAmt < 0.05) {
   630→      // kill all
   631→      for (let i = 0; i < this.sprayCount; i++) this.sprayLife[i] = 0;
   632→      (this.spray.geometry.attributes.aLife as THREE.BufferAttribute).needsUpdate = true;
   633→      return;
   634→    }
   635→    // emit from player rear wheel
   636→    const rate = clamp(bike.v / 30, 0, 1) * 90 * rainAmt;
   637→    let n = Math.floor(rate * dt + (this.rng.next() < (rate * dt) % 1 ? 1 : 0));
   638→    while (n-- > 0) this.emitSpray(bike.worldPos.x + (this.rng.next() - 0.5) * 0.5, bike.worldPos.y + 0.15, bike.worldPos.z, -bike.v * 0.3);
   639→    // emit from traffic rear axles
   640→    if (this.frameIdx % 2 === 0) {
   641→      for (const p of this._sprayPts) {
   642→        if (this.rng.next() < 0.5) this.emitSpray(p.x, p.y, p.z, -8);
   643→      }
   644→    }
   645→    // integrate
   646→    for (let i = 0; i < this.sprayCount; i++) {
   647→      if (this.sprayLife[i] <= 0) continue;
   648→      this.sprayLife[i] -= dt * 1.6;
   649→      this.sprayPos[i * 3] += this.sprayVel[i * 3] * dt;
   650→      this.sprayPos[i * 3 + 1] += this.sprayVel[i * 3 + 1] * dt;
   651→      this.sprayPos[i * 3 + 2] += this.sprayVel[i * 3 + 2] * dt;
   652→      this.sprayVel[i * 3 + 1] += 2.2 * dt; // mist rises slightly then
   653→      if (this.sprayLife[i] < 0) this.sprayLife[i] = 0;
   654→    }
   655→    (this.spray.geometry.attributes.position as THREE.BufferAttribute).needsUpdate = true;
   656→    (this.spray.geometry.attributes.aLife as THREE.BufferAttribute).needsUpdate = true;
   657→    void highway;
   658→    void traffic;
   659→  }
   660→
   661→  private _sprayPts: THREE.Vector3[] = [];
   662→  frameIdx = 0;
   663→
   664→  emitSpray(x: number, y: number, z: number, backVel: number) {
   665→    const i = this.sprayCursor;
   666→    this.sprayCursor = (this.sprayCursor + 1) % this.sprayCount;
   667→    this.sprayPos[i * 3] = x;
   668→    this.sprayPos[i * 3 + 1] = y;
   669→    this.sprayPos[i * 3 + 2] = z;
   670→    this.sprayVel[i * 3] = (this.rng.next() - 0.5) * 1.4;
   671→    this.sprayVel[i * 3 + 1] = this.rng.range(0.5, 2.0);
   672→    this.sprayVel[i * 3 + 2] = backVel * (0.5 + this.rng.next() * 0.5) + (this.rng.next() - 0.5);
   673→    this.sprayLife[i] = 1;
   674→  }

   681→  private refreshEnv() {
   682→    if (!this.envEnabled) return;
   683→    const old = this.envRT;

   689→  private applyAll() {
   690→    const c = this.current;
   691→    const u = this.skyMat.uniforms;
   692→    (u.zenith.value as THREE.Color).copy(c.zenith);
   693→    (u.mid.value as THREE.Color).copy(c.mid);
   694→    (u.horizon.value as THREE.Color).copy(c.horizon);
   695→    (u.sunDir.value as THREE.Vector3).copy(c.sunDir).normalize();
   696→    (u.sunColor.value as THREE.Color).copy(c.sunColor);
   697→    u.sunDisc.value = c.sunDisc;
   698→    u.sunGlow.value = c.sunGlow;
   699→    u.stars.value = c.starIntensity;
   700→    (u.cloudTint.value as THREE.Color).copy(c.cloudTint);
   701→
   702→    const scene = this.scene;
   703→    if (scene.fog instanceof THREE.FogExp2) {
   704→      scene.fog.color.copy(c.fogColor);
   705→      scene.fog.density = c.fogDensity;
   706→    }
   707→
   708→    this.sun.color.copy(c.sunLightColor);
   709→    this.sun.intensity = c.sunLightIntensity;
   710→    this.ambient.color.copy(c.ambientColor);
   711→    this.ambient.intensity = c.ambientIntensity;
   712→    this.sun.castShadow = c.sunLightIntensity > 0.5;
   713→
   714→    // shared highway materials
   715→    this.mats.lampCone.opacity = c.lampCone;
   716→    this.mats.lampPool.opacity = c.lampPool;
   717→    this.mats.lampHead.color.setRGB(
   718→      clamp(0.55 + c.windowEmissive * 0.45, 0, 1.6),
   719→      clamp(0.42 + c.windowEmissive * 0.22, 0, 1.2),
   720→      clamp(0.28 + c.windowEmissive * 0.1, 0, 0.9)
   721→    );
   722→    for (const w of this.mats.windows) w.emissiveIntensity = c.windowEmissive;
   723→    this.mats.sign.emissiveIntensity = c.signEmissive;
   724→    this.mats.concrete.color.setScalar(lerp(0.74, 0.4, clamp(c.fogDensity / 0.003, 0, 1)));
   725→  }

   769→function blendParams(cur: WeatherParams, target: WeatherParams, t: number) {
   770→  const e = t * t * (3 - 2 * t);
   771→  cur.zenith.lerp(target.zenith, e);
   772→  cur.mid.lerp(target.mid, e);
   773→  cur.horizon.lerp(target.horizon, e);
   774→  cur.sunDir.lerp(target.sunDir, e);
   775→  cur.sunColor.lerp(target.sunColor, e);
   776→  cur.cloudTint.lerp(target.cloudTint, e);
   777→  cur.fogColor.lerp(target.fogColor, e);
   778→  cur.sunLightColor.lerp(target.sunLightColor, e);
   779→  cur.ambientColor.lerp(target.ambientColor, e);
   780→  cur.sunDisc = lerp(cur.sunDisc, target.sunDisc, e);
   781→  cur.sunGlow = lerp(cur.sunGlow, target.sunGlow, e);
   782→  cur.starIntensity = lerp(cur.starIntensity, target.starIntensity, e);
   783→  cur.fogDensity = lerp(cur.fogDensity, target.fogDensity, e);
   784→  cur.sunLightIntensity = lerp(cur.sunLightIntensity, target.sunLightIntensity, e);
   785→  cur.ambientIntensity = lerp(cur.ambientIntensity, target.ambientIntensity, e);
   786→  cur.lampCone = lerp(cur.lampCone, target.lampCone, e);
   787→  cur.lampPool = lerp(cur.lampPool, target.lampPool, e);
   788→  cur.windowEmissive = lerp(cur.windowEmissive, target.windowEmissive, e);
   789→  cur.signEmissive = lerp(cur.signEmissive, target.signEmissive, e);
   790→  cur.bloom = lerp(cur.bloom, target.bloom, e);
   791→  cur.exposure = lerp(cur.exposure, target.exposure, e);
   792→  cur.saturation = lerp(cur.saturation, target.saturation, e);
   793→  cur.contrast = lerp(cur.contrast, target.contrast, e);
   794→  cur.rain = lerp(cur.rain, target.rain, e);
   795→  cur.wetness = lerp(cur.wetness, target.wetness, e);
   796→  cur.embers = lerp(cur.embers, target.embers, e);
   797→  cur.headlights = lerp(cur.headlights, target.headlights, e);
   798→  cur.headlightCones = target.headlightCones && e > 0.5 ? true : cur.headlightCones;
   799→  if (e > 0.999) cur.headlightCones = target.headlightCones;
   800→  cur.glare = lerp(cur.glare, target.glare, e);
   801→  cur.vignette = lerp(cur.vignette, target.vignette, e);
   802→}
